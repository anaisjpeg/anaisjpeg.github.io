// ============================================================ // ADIL 57 – Apps Script v3 // Architecture : // "Pointage_Prénom" ← une feuille par collaborateur // "Récup_Prénom" ← contrepartie/récup par collaborateur // "HS_HC_Prénom" ← résultats (menu direction) // ============================================================ const CONFIG = { SEUIL_REPOS_MIN: 7 * 60, // Pause midi : déduite si la plage chevauche 12h30–13h30 PAUSE_DEBUT_H: 12, PAUSE_DEBUT_M: 30, PAUSE_FIN_H: 13, PAUSE_FIN_M: 30, PAUSE_DUREE_MIN: 60, CONTRATS: { "Anaïs": 35, "Cloé": 35, "Delphine": 28, "Elsa": 28, "Frédéric": 35, "Julia": 28, "Mélissa": 28, "Thibaut": 35, "Thomas": 35, "Valentin": 35, "Véronique": 35, }, }; // ── doGet ──────────────────────────────────────────────────── function doGet(e) { const action = e.parameter.action || "pointage"; if (action === "pointage") return handlePointage(e); if (action === "recup") return handleRecup(e); if (action === "getSemaines") return handleGetSemaines(e); if (action === "getSolde") return handleGetSolde(e); return json({ ok: false, error: "Action inconnue" }); } // ── ACTION : pointage ──────────────────────────────────────── function handlePointage(e) { const nom = (e.parameter.nom || "").trim(); const type = (e.parameter.type || "").trim(); const commentaire = (e.parameter.commentaire || "").trim(); if (!nom || !type) return json({ ok: false, error: "Missing parameters" }); const sheet = getOrCreateSheet("Pointage_" + nom, ["Prénom", "Date et heure", "Type", "Commentaire"]); sheet.appendRow([nom, new Date(), type, commentaire]); return json({ ok: true }); } // ── ACTION : saisie récup/contrepartie ─────────────────────── function handleRecup(e) { const nom = (e.parameter.nom || "").trim(); const lundiISO = (e.parameter.semaine || "").trim(); const contrepartie = (e.parameter.contrepartie || "").trim(); const recupMin = parseInt(e.parameter.recupMin || "0", 10); const commentaire = (e.parameter.commentaire || "").trim(); if (!nom || !lundiISO) return json({ ok: false, error: "Missing parameters" }); const sheet = getOrCreateSheet("Récup_" + nom, ["Prénom", "Lundi (ISO)", "Contrepartie", "Récup prise (min)", "Date saisie", "Commentaire"]); const data = sheet.getDataRange().getValues(); let rowIdx = -1; for (let i = 1; i < data.length; i++) { if (String(data[i][1]).trim() === lundiISO) { rowIdx = i + 1; break; } } if (rowIdx > 0) { sheet.getRange(rowIdx, 3, 1, 4).setValues([[contrepartie, recupMin, new Date(), commentaire]]); } else { sheet.appendRow([nom, lundiISO, contrepartie, recupMin, new Date(), commentaire]); } return json({ ok: true }); } // ── ACTION : getSemaines (temps réel) ──────────────────────── function handleGetSemaines(e) { const nom = (e.parameter.nom || "").trim(); if (!nom) return json({ ok: false, error: "Missing nom" }); const resultats = calculerPourCollaborateur(nom); return json({ ok: true, semaines: resultats.map(r => ({ semaine: r.semaineTxt, lundi: r.lundiISO, totalH: r.totalEffectifH, depassementH: r.depassementH, depassementMin: r.depassementMin, type: r.typeCalc, contrepartie: r.contrepartie, reposAcquisH: r.reposAcquisH, reposAcquisMin: r.reposAcquisMin, alerte: r.alerte, evenements: r.evenements, }))}); } // ── ACTION : getSolde (temps réel) ──────────────────────────── function handleGetSolde(e) { const nom = (e.parameter.nom || "").trim(); if (!nom) return json({ ok: false, error: "Missing nom" }); const resultats = calculerPourCollaborateur(nom); const soldeMin = resultats.reduce((sum, r) => { if (r.contrepartie === "Paiement") return sum; return sum + r.reposAcquisMin; }, 0); return json({ ok: true, soldeMin, soldeH: minutesEnHHMM(soldeMin), alerte: soldeMin >= CONFIG.SEUIL_REPOS_MIN }); } // ── CALCUL CENTRAL ──────────────────────────────────────────── function calculerPourCollaborateur(nom) { const contratH = CONFIG.CONTRATS[nom] || 35; const contratMin = contratH * 60; const tempsPlein = contratH >= 35; const pSheet = SpreadsheetApp.getActiveSpreadsheet() .getSheetByName("Pointage_" + nom); if (!pSheet) return []; // Lire pointages const rawData = pSheet.getDataRange().getValues(); const pointages = []; for (let i = 1; i < rawData.length; i++) { const dt = rawData[i][1]; const type = String(rawData[i][2]).trim().toLowerCase(); const comment = String(rawData[i][3] || "").trim(); if (!dt || (type !== "debut" && type !== "fin")) continue; const d = dt instanceof Date ? dt : new Date(dt); if (isNaN(d)) continue; pointages.push({ date: d, type, comment }); } pointages.sort((a, b) => a.date - b.date); // Lire récup/contrepartie const recupMap = lireRecupMap(nom); // Grouper par semaine const groupes = {}; for (const p of pointages) { const lundi = getLundi(p.date); const iso = formatDateISO(lundi); if (!groupes[iso]) groupes[iso] = { lundi, pointages: [] }; groupes[iso].pointages.push(p); } const resultats = []; for (const lundiISO of Object.keys(groupes).sort().reverse()) { const { lundi, pointages: pts } = groupes[lundiISO]; const dimanche = new Date(lundi); dimanche.setDate(dimanche.getDate() + 6); const plages = apparierPlages(pts); const totalBrutMin = plages.reduce((s, p) => s + p.dureeMin, 0); const recupEntry = recupMap[lundiISO] || {}; const recupPriseMin = recupEntry.recupMin || 0; const contrepartie = recupEntry.contrepartie || ""; // Temps effectif = brut − récup prises const totalEffectif = Math.max(0, totalBrutMin - recupPriseMin); const depassementMin = Math.max(0, totalEffectif - contratMin); let t1 = 0, t2 = 0, reposMin = 0, alerte35 = false; if (depassementMin > 0) { if (tempsPlein) { t1 = Math.min(depassementMin, 8 * 60); t2 = Math.max(0, depassementMin - 8 * 60); reposMin = Math.round(t1 * 1.25 + t2 * 1.5); } else { const lim = contratMin / 10; t1 = Math.min(depassementMin, lim); t2 = Math.max(0, depassementMin - lim); reposMin = Math.round(t1 * 1.10 + t2 * 1.25); if (totalEffectif > 35 * 60) alerte35 = true; } } // Si paiement : repos comptabilisé = 0 const reposComptabilise = contrepartie === "Paiement" ? 0 : reposMin; let alerte = ""; if (alerte35) alerte = "⚠️ Dépasse 35h (temps partiel)"; else if (depassementMin > 0 && !contrepartie) alerte = "Contrepartie non renseignée"; // Événements lisibles const evenements = plages.map(p => ({ date: formatDateFR(p.debut), heure: formatHeure(p.debut) + "–" + formatHeure(p.fin), dureeH: minutesEnHHMM(p.dureeMin), pauseDeduite: p.pauseDeduite, commentaire: p.comment || "", })); resultats.push({ lundiISO, semaineTxt: "S" + getNumSemaine(lundi) + " · " + formatDateFR(lundi) + " – " + formatDateFR(dimanche), contratH, totalEffectifH: minutesEnHHMM(totalEffectif), totalEffectifMin: totalEffectif, depassementH: minutesEnHHMM(depassementMin), depassementMin, typeCalc: depassementMin > 0 ? (tempsPlein ? "HS" : "HC") : "—", tranche1H: minutesEnHHMM(t1), tranche2H: minutesEnHHMM(t2), reposAcquisH: minutesEnHHMM(reposComptabilise), reposAcquisMin: reposComptabilise, contrepartie, recupPriseH: minutesEnHHMM(recupPriseMin), alerte, evenements, }); } return resultats; } // ── APPARIEMENT début/fin + déduction pause midi ────────────── // La pause 12h30–13h30 (1h) est déduite automatiquement // si la plage chevauche cette fenêtre d'au moins 30 min function apparierPlages(pointages) { const plages = []; let debut = null; for (const p of pointages) { if (p.type === "debut") { debut = p; } else if (p.type === "fin" && debut) { const brutMin = Math.round((p.date - debut.date) / 60000); if (brutMin > 0) { const { netMin, pauseDeduite } = deduirePauseMidi(debut.date, p.date, brutMin); plages.push({ debut: debut.date, fin: p.date, dureeMin: netMin, pauseDeduite, comment: debut.comment || p.comment, }); } debut = null; } } return plages; } // Déduit 1h si la plage couvre 12h30–13h30 (chevauchement ≥ 30 min) function deduirePauseMidi(dateDebut, dateFin, brutMin) { const pauseDebut = new Date(dateDebut); pauseDebut.setHours(CONFIG.PAUSE_DEBUT_H, CONFIG.PAUSE_DEBUT_M, 0, 0); const pauseFin = new Date(dateDebut); pauseFin.setHours(CONFIG.PAUSE_FIN_H, CONFIG.PAUSE_FIN_M, 0, 0); const chevauchDebut = Math.max(dateDebut, pauseDebut); const chevauchFin = Math.min(dateFin, pauseFin); const chevauchMin = Math.max(0, Math.round((chevauchFin - chevauchDebut) / 60000)); // Déduire seulement si le chevauchement est ≥ 30 min // (évite de pénaliser une plage courte qui touche juste midi) if (chevauchMin >= 30) { return { netMin: brutMin - CONFIG.PAUSE_DUREE_MIN, pauseDeduite: true }; } return { netMin: brutMin, pauseDeduite: false }; } // ── MENU DIRECTION : calcul + écriture feuilles HS_HC_Prénom ── function calculerTousLesCollaborateurs() { const ui = SpreadsheetApp.getUi(); const prenoms = Object.keys(CONFIG.CONTRATS); let total = 0; const alertes = []; for (const nom of prenoms) { const resultats = calculerPourCollaborateur(nom); if (resultats.length === 0) continue; ecrireFeuilleCollaborateur(nom, resultats); total += resultats.length; const solde = resultats.reduce((s, r) => r.contrepartie === "Paiement" ? s : s + r.reposAcquisMin, 0); if (solde >= CONFIG.SEUIL_REPOS_MIN) alertes.push(nom + " : " + minutesEnHHMM(solde) + " à poser"); } let msg = "✅ " + total + " ligne(s) pour " + prenoms.length + " collaborateur(s)."; if (alertes.length) msg += "\n\n⚠️ SEUIL 7H ATTEINT :\n" + alertes.join("\n"); ui.alert(msg); } function ecrireFeuilleCollaborateur(nom, resultats) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheetName = "HS_HC_" + nom; let sheet = ss.getSheetByName(sheetName); if (!sheet) sheet = ss.insertSheet(sheetName); sheet.clearContents(); sheet.clearFormats(); const headers = ["Semaine","Contrat (h)","Total effectif","Dépassement", "Type","Tranche 1","Tranche 2","Repos acquis","Récup prises","Contrepartie","Alerte","Détail"]; sheet.getRange(1,1,1,headers.length).setValues([headers]) .setFontWeight("bold").setBackground("#1a252f").setFontColor("#ffffff").setHorizontalAlignment("center"); sheet.setFrozenRows(1); const rows = resultats.map(r => [ r.semaineTxt, r.contratH, r.totalEffectifH, r.depassementH, r.typeCalc, r.tranche1H, r.tranche2H, r.reposAcquisH, r.recupPriseH, r.contrepartie || "—", r.alerte, r.evenements.map(e => e.date + " " + e.heure + (e.pauseDeduite ? " (-1h pause)" : "") + (e.commentaire ? " [" + e.commentaire + "]" : "") ).join(" | "), ]); sheet.getRange(2,1,rows.length,headers.length).setValues(rows); for (let i = 0; i < resultats.length; i++) { const r = resultats[i]; const range = sheet.getRange(i+2,1,1,headers.length); if (r.alerte.includes("⚠️")) range.setBackground("#f39c12"); else if (r.alerte) range.setBackground("#fdebd0"); else if (r.depassementMin > 0) range.setBackground("#fef9e7"); else range.setBackground(i%2===0 ? "#f4f6f7" : "#ffffff"); } [220,80,110,110,60,100,100,110,100,110,200,500].forEach((w,i) => sheet.setColumnWidth(i+1,w)); } // ── UTILITAIRES ─────────────────────────────────────────────── function lireRecupMap(nom) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Récup_" + nom); const map = {}; if (!sheet) return map; const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { const iso = String(data[i][1]).trim(); if (!iso) continue; map[iso] = { contrepartie: String(data[i][2]).trim(), recupMin: parseInt(data[i][3] || "0", 10), }; } return map; } function getOrCreateSheet(name, headers) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); sheet.getRange(1,1,1,headers.length).setValues([headers]) .setFontWeight("bold").setBackground("#1a252f").setFontColor("#ffffff"); sheet.setFrozenRows(1); } return sheet; } function getLundi(date) { const d = new Date(date); const day = d.getDay(); d.setDate(d.getDate() + (day === 0 ? -6 : 1 - day)); d.setHours(0,0,0,0); return d; } function getNumSemaine(date) { const d = new Date(Date.UTC(date.getFullYear(), date.getMonth(), date.getDate())); const dayNum = d.getUTCDay() || 7; d.setUTCDate(d.getUTCDate() + 4 - dayNum); const yearStart = new Date(Date.UTC(d.getUTCFullYear(), 0, 1)); return Math.ceil((((d - yearStart) / 86400000) + 1) / 7); } function formatDateISO(d) { return d.getFullYear()+"-"+String(d.getMonth()+1).padStart(2,"0")+"-"+String(d.getDate()).padStart(2,"0"); } function formatDateFR(d) { return String(d.getDate()).padStart(2,"0")+"/"+String(d.getMonth()+1).padStart(2,"0")+"/"+d.getFullYear(); } function formatHeure(d) { return String(d.getHours()).padStart(2,"0")+":"+String(d.getMinutes()).padStart(2,"0"); } function minutesEnHHMM(min) { if (!min || min <= 0) return "0h00"; return Math.floor(min/60)+"h"+String(Math.round(min%60)).padStart(2,"0"); } function json(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } function onOpen() { SpreadsheetApp.getUi().createMenu("⏱ ADIL 57") .addItem("Recalculer tous les collaborateurs","calculerTousLesCollaborateurs") .addToUi(); }
